create table games (
  
id nvarchar(36),
  
league nvarchar(100),
  
competitor1 nvarchar(100),
  
competitor2 nvarchar(100),
  
gametimestamp datetime,
  
tvstations nvarchar(100),
  
primary key (id)
);
