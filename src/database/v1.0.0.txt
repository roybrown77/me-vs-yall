create table game (
  
id nvarchar(36) not null,

league nvarchar(100) not null,

competitor1 nvarchar(100) not null,

competitor2 nvarchar(100) not null,

winner nvarchar(100),

gametimestamp datetime not null,

tvstation nvarchar(100) not null,

primary key (id)
);

create table user (

id nvarchar(36) not null,

email nvarchar(100) not null,

password nvarchar(100) not null,

primary key (id)
);

id nvarchar(36) not null,

name nvarchar(100) not null,

userid nvarchar(36) not null,

primary key (id)
